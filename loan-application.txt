                        <input type="number" id="gold_weight" name="gold_weight" required>

                        <input type="number" id="gold_purity" name="gold_purity" required>
                    </div>
                </div>
                <div class="row">
                    <div class="col">
                        <label for="gold_value">Estimated Value (₹)</label>
                        <input type="number" id="gold_value" name="gold_value" required>
                    </div>
                </div>
            `;
        } else if(type === 'home'){
            html = `
                <div class="row">
                    <div class="col">
                        <label for="property_address">Property Address</label>
                        <input type="text" id="property_address" name="property_address" required>
                    </div>
                    <div class="col">
                        <label for="co_applicant">Co-Applicant Name (optional)</label>
                        <input type="text" id="co_applicant" name="co_applicant">
                    </div>
                </div>
            `;
        } else if(type === 'car'){
            html = `
                <div class="row">
                    <div class="col">
                        <label for="vehicle_model">Vehicle Model</label>
                        <input type="text" id="vehicle_model" name="vehicle_model" required>
                    </div>
                    <div class="col">
                        <label for="dealer_info">Dealer Info</label>
                        <input type="text" id="dealer_info" name="dealer_info">
                    </div>
                </div>
            `;
        } else if(type === 'education'){
            html = `
                <div class="row">
                    <div class="col">
                        <label for="college_name">College / University</label>
                        <input type="text" id="college_name" name="college_name" required>
                    </div>
                    <div class="col">
                        <label for="course_name">Course</label>
                        <input type="text" id="course_name" name="course_name" required>
                    </div>
                </div>
                <div class="row">
                    <div class="col">
                        <label for="tuition_fee">Tuition Fee (₹)</label>

            html = `<p>No extra details required for Personal Loan.</p>`;
        }
        loanSpecificDiv.innerHTML = html;
    }

        renderLoanSpecificFields(e.target.value);
    });

    renderLoanSpecificFields(loanTypeSelect.value);
